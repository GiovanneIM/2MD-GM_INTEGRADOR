function abrirModal(idModal) {
    const modal = new bootstrap.Modal(document.getElementById(idModal));
    modal.show();
}

function fecharModal(idModal) {
    const modal = bootstrap.Modal.getInstance(document.getElementById(idModal));
    modal.hide();
}

function toggleModal(idModal, action) {
    const element = document.getElementById(idModal);
    const modal = bootstrap.Modal.getInstance(element) || new bootstrap.Modal(element);

    modal[action]();
}
